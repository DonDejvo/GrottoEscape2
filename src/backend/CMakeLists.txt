file(GLOB BACKEND_COMMON CONFIGURE_DEPENDS *.cpp)
add_library(backend STATIC ${BACKEND_COMMON})
target_include_directories(backend PUBLIC ${CMAKE_SOURCE_DIR}/src)
target_link_libraries(backend PUBLIC assets glm::glm)

if(USE_GLES)
    file(GLOB GLES_SOURCES CONFIGURE_DEPENDS gles/*.cpp)
    target_sources(backend PRIVATE ${GLES_SOURCES})
    target_compile_definitions(backend PUBLIC BACKEND_GLES)
    target_link_options(backend PUBLIC
        "-sUSE_GLFW=3" "-sUSE_WEBGL2=1" "-sFULL_ES3=1")
else()
    file(GLOB WEBGPU_SOURCES CONFIGURE_DEPENDS webgpu/*.cpp)
    target_sources(backend PRIVATE ${WEBGPU_SOURCES})
    target_compile_definitions(backend PUBLIC BACKEND_WEBGPU)
    if(EMSCRIPTEN)
        target_link_libraries(backend PUBLIC emdawnwebgpu_cpp webgpu_glfw)
        target_link_options(backend PUBLIC "-sUSE_GLFW=3" "-sASYNCIFY=1")
    else()
        target_link_libraries(backend PUBLIC webgpu_dawn webgpu_glfw glfw)
    endif()
endif()
